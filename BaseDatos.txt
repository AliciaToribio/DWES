CREATE TABLE empleados(
dni VARCHAR(100) PRIMARY KEY,
nombre VARCHAR(100)NOT NULL,
sexo CHAR(1)NOT NULL,
categoria INT(11),
anyos INT(6)
); 

CREATE TABLE nominas(
id_nomina INT(11) PRIMARY KEY AUTO_INCREMENT,
sueldo DECIMAL(10,2) NOT NULL,
empleado VARCHAR(100) NOT NULL UNIQUE,
CONSTRAINT fk_empleado FOREIGN KEY(empleado) REFERENCES empleados(dni) ON DELETE CASCADE ON UPDATE CASCADE
);
